\section{Examples}
    The following examples demonstrate the process of breaking down Hangul syllables into their constituent parts, allowing learners to understand the structure and pronunciation of Korean words.

    \begin{align}
        \text{가} &= 
        \underbrace{
        \begin{pmatrix}
            \text{ㄱ}\\
            \boxed{\mathrm{c}_1}\\
            \mathrm{g}
        \end{pmatrix}
        }_{\mathrm{c}_i}
        +
        \underbrace{
        \begin{pmatrix}
            \text{}\\
            \boxed{\mathrm{v}_1}\\
            \mathrm{}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{ㅏ}\\
            \boxed{\mathrm{v}_2}\\
            \mathrm{a}
        \end{pmatrix}
        }_{\mathrm{v}_t}
        +
        \underbrace{
        \begin{pmatrix}
            \text{}\\
            \boxed{\mathrm{c}_2}\\
            \mathrm{}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{}\\
            \boxed{\mathrm{c}_3}\\
            \mathrm{}
        \end{pmatrix}
        }_{\mathrm{c}_o}
        \\&=
        \begin{pmatrix}
            \text{ㄱ}\\
            \boxed{\mathrm{c}_i}\\
            \mathrm{g}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{ㅏ}\\
            \boxed{\mathrm{v}_t}\\
            \mathrm{a}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{}\\
            \boxed{\mathrm{c}_o}\\
            \mathrm{}
        \end{pmatrix}
        =
        \begin{pmatrix}
            \text{가}\\
            \boxed{\mathrm{syllable}}\\
            \mathrm{ga}
        \end{pmatrix}
    \end{align}
    \\

    \begin{align}
        \text{왔} &= 
        \underbrace{
        \begin{pmatrix}
            \text{ㅇ}\\
            \boxed{\mathrm{c}_1}\\
            \varnothing
        \end{pmatrix}
        }_{\mathrm{c}_i}
        +
        \underbrace{
        \begin{pmatrix}
            \text{ㅗ}\\
            \boxed{\mathrm{v}_1}\\
            \mathrm{o}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{ㅏ}\\
            \boxed{\mathrm{v}_2}\\
            \mathrm{a}
        \end{pmatrix}
        }_{\mathrm{v}_t}
        +
        \underbrace{
        \begin{pmatrix}
            \text{ㅅ}\\
            \boxed{\mathrm{c}_2}\\
            \mathrm{d}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{ㅅ}\\
            \boxed{\mathrm{c}_3}\\
            \mathrm{d}
        \end{pmatrix}
        }_{\mathrm{c}_o}
        \\&=
        \begin{pmatrix}
            \text{ㅇ}\\
            \boxed{\mathrm{c}_i}\\
            \varnothing
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{ㅘ}\\
            \boxed{\mathrm{v}_t}\\
            \mathrm{wa}
        \end{pmatrix}
        +
        \begin{pmatrix}
            \text{ㅆ}\\
            \boxed{\mathrm{c}_o}\\
            \mathrm{d}
        \end{pmatrix}
        =
        \begin{pmatrix}
            \text{왔}\\
            \boxed{\mathrm{syllable}}\\
            \mathrm{wad}
        \end{pmatrix}
    \end{align}
